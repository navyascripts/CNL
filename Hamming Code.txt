#include <stdio.h>
#include <math.h>

int main()
{
    int data[100];
    int code[200];

    int n, r = 0;
    int total;
    int i, j;
    int parity;
    int errorPosition;
    int error = 0;

    printf("Enter number of data bits: ");
    scanf("%d", &n);

    printf("Enter data bits: ");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &data[i]);
    }

    while((1 << r) < (n + r + 1))
    {
        r++;
    }

    total = n + r;

    printf("\nNumber of data bits  : %d", n);
    printf("\nNumber of parity bits: %d", r);
    printf("\nTotal bits           : %d\n", total);
  for(i = 1; i <= total; i++)
    {
        code[i] = 0;
    }
    j = 0;

    for(i = 1; i <= total; i++)
    {
        if((i & (i - 1)) != 0)
        {
            code[i] = data[j];
            j++;
        }
    }
for(i = 0; i < r; i++)
    {
        int position = 1 << i;

        parity = 0;

        for(j = 1; j <= total; j++)
        {
            if(j & position)
            {
                parity = parity ^ code[j];
            }
        }

        code[position] = parity;
    }
  printf("\nHamming Code: ");

    for(i = total; i >= 1; i--)
    {
        printf("%d", code[i]);
    }
    printf("\n\nEnter error position (0 for no error): ");
    scanf("%d", &errorPosition);

    if(errorPosition != 0)
    {
        if(errorPosition >= 1 && errorPosition <= total)
        {
            code[errorPosition] = code[errorPosition] ^ 1;

            printf("\nError introduced at position %d.\n",
                   errorPosition);
        }
        else
        {
            printf("\nInvalid error position.\n");
            return 0;
        }
    }

    error = 0;

    for(i = 0; i < r; i++)
    {
        int position = 1 << i;

        parity = 0;

        for(j = 1; j <= total; j++)
        {
            if(j & position)
            {
                parity = parity ^ code[j];
            }
        }

        if(parity != 0)
        {
            error = error + position;
        }
    }

    if(error == 0)
    {
        printf("No error detected.\n");
    }
    else
    {
        printf("Error detected at position: %d\n",
               error);

        /* Correct the error */
        code[error] = code[error] ^ 1;

        printf("Error corrected successfully.\n");
    }

    printf("Corrected Hamming Code: ");

    for(i = total; i >= 1; i--)
    {
        printf("%d", code[i]);
    }

    printf("\n");

    return 0;
}
